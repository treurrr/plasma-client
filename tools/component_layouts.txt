# Wire layout of every data component, read by tools/gen_components.py, and
# of the other bodies that are easiest to say the same way: particles, recipe
# displays and the packets in session_data.cpp that are only data.
#
# From the protocol wiki's Slot data page as it stood for 1.21.10 and 1.21.11,
# and checked against bytes the 1.21.11 server sent (tests/test_data.cpp says
# which). Where the wiki and the bytes disagree the bytes win.
#
#   varint varlong byte ubyte short int long float double bool string
#   identifier nbt component uuid position slot     one wire value
#   optvarint                a varint: 0 is none, else the value plus one
#   bytes                    a varint length, then that many bytes
#   -                        nothing at all
#   {name:L name:L}          the fields in order
#   list(L)                  a varint count, then that many
#   opt(L)                   a bool, then L when it is true
#   enum(a,b,c=99)           a varint naming one of these
#   id(registry)             a varint id in that registry
#   idor(registry, L)        a varint: 0 means L follows, else the id plus one
#   idset(registry)          a varint: 0 means a tag name follows, else n-1 ids
#   oneof(a:L, b:L)          a varint picking a case by position; a bool tag
#                            is the same byte, so oneof(false:L, true:L)
#   switch(registry){a:L}    a varint id in that registry picks the case by
#                            its name, without minecraft:; an entry not
#                            listed carries nothing
#   typed_component          a component type id, then that component's body
#   @name                    one of the named types below

type property = {name:string match:oneof(range:{min:opt(string) max:opt(string)}, exact:{value:string})}
type block_predicate = {blocks:opt(idset(minecraft:block)) properties:opt(list(@property)) nbt:opt(nbt) components:list(typed_component) partial:list({type:oneof(component:id(minecraft:data_component_type), predicate:id(minecraft:data_component_predicate_type)) predicate:nbt})}

type kinetic_condition = {max_duration:varint min_speed:float min_relative_speed:float}
type filtered_string = {raw:string filtered:opt(string)}
type filtered_component = {raw:component filtered:opt(component)}
type particle = switch(minecraft:particle_type){block:{state:varint} block_marker:{state:varint} dragon_breath:{power:float} dust:{color:int scale:float} dust_color_transition:{from:int to:int scale:float} effect:{color:int power:float} entity_effect:{color:int} falling_dust:{state:varint} tinted_leaves:{color:int} sculk_charge:{roll:float} flash:{color:int} instant_effect:{color:int power:float} item:{item:slot} vibration:{source:switch(minecraft:position_source_type){block:{pos:position} entity:{entity:varint eye_height:float}} ticks:varint} trail:{x:double y:double z:double color:int duration:varint} shriek:{delay:varint} dust_pillar:{state:varint} block_crumble:{state:varint}}

type slot_display = switch(minecraft:slot_display){item:{item:id(minecraft:item)} item_stack:{stack:slot} tag:{tag:identifier} smithing_trim:{base:@slot_display material:@slot_display pattern:idor(minecraft:trim_pattern, @trim_pattern)} with_remainder:{input:@slot_display remainder:@slot_display} composite:{options:list(@slot_display)}}
type recipe_display = switch(minecraft:recipe_display){crafting_shapeless:{ingredients:list(@slot_display) result:@slot_display station:@slot_display} crafting_shaped:{width:varint height:varint ingredients:list(@slot_display) result:@slot_display station:@slot_display} furnace:{ingredient:@slot_display fuel:@slot_display result:@slot_display station:@slot_display time:varint experience:float} stonecutter:{ingredient:@slot_display result:@slot_display station:@slot_display} smithing:{template:@slot_display base:@slot_display addition:@slot_display result:@slot_display station:@slot_display}}
